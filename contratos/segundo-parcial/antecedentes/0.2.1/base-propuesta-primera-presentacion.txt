-- En el proyecto real integrar mediante Alembic, backfill y pruebas de regresion.
-- No insertar manualmente una revision ficticia en public.alembic_version.
--
-- AJUSTES PRINCIPALES
-- 1. empresa_id en detalles, asociaciones y registros empresariales que lo omitian.
-- 2. FKs compuestas para impedir referencias cruzadas entre empresas.
-- 3. Correos unicos por empresa; cuentas de plataforma con ambito separado.
-- 4. SKU, codigos y referencias de negocio unicos dentro de cada empresa.
-- 5. CHECKs para importes, cantidades, fechas, alcance y sincronizacion.
-- 6. Origen temporal obligatorio y plazo de reserva 24/72 h desde ese origen.
-- 7. Esquemas y ENUMs incluidos; planes iniciales 2/10 configurables.
--
-- LECTURA PARA EL DIAGRAMA
-- id sigue siendo PK global UUID. UNIQUE(id,empresa_id) permite FKs de tenant.
-- empresa_id repetido en detalles es redundancia controlada de integridad.
-- Una columna de FK nullable representa una relacion opcional.
-- clientes y empleados usan usuario_id como PK y FK (especializaciones).
-- ciudades, roles y planes son referencias globales; empresas son los tenants.
-- El administrador inicial se representa por usuario EMPRESA + rol autorizado;
-- el alta consistente de empresa y administrador se realiza desde el backend.
-- Alembic_version es metadato tecnico y puede omitirse del diagrama de negocio.

(
    id uuid NOT NULL DEFAULT gen_random_uuid(),

    usuario_id uuid NOT NULL,
    operacion_id uuid NOT NULL,
    tipo character varying(40) COLLATE pg_catalog."default" NOT NULL,
    version_esquema integer NOT NULL,
    dispositivo_id uuid NOT NULL,
    secuencia_local bigint NOT NULL,
    creada_local_en timestamp with time zone NOT NULL,
    recibida_en timestamp with time zone NOT NULL,
    aceptada_en timestamp with time zone,
    referencia_horaria_id uuid,
    hash_solicitud character varying(64) COLLATE pg_catalog."default" NOT NULL,
    solicitud jsonb NOT NULL,
    estado character varying(40) COLLATE pg_catalog."default" NOT NULL,
    codigo character varying(40) COLLATE pg_catalog."default" NOT NULL,
    detalle text COLLATE pg_catalog."default" NOT NULL,
    resultado jsonb NOT NULL,
    recurso_id uuid,
    revision integer NOT NULL DEFAULT 1,
    CONSTRAINT operaciones_sincronizacion_pkey PRIMARY KEY (id),
    CONSTRAINT uq_operaciones_actor_uuid UNIQUE (empresa_id, usuario_id, operacion_id),
    CONSTRAINT uq_operaciones_id_empresa UNIQUE (id, empresa_id),
    CONSTRAINT ck_operaciones_version_secuencia CHECK (version_esquema >= 1 AND secuencia_local >= 1 AND revision >= 1),
    CONSTRAINT ck_operaciones_solicitud_objeto CHECK (jsonb_typeof(solicitud) = 'object' AND jsonb_typeof(resultado) = 'object'),
    CONSTRAINT ck_operaciones_hash CHECK (hash_solicitud ~ '^[0-9a-f]{64}$'),
    CONSTRAINT ck_operaciones_tipo CHECK (tipo IN ('SOLICITAR_RESERVA','SOLICITAR_COMPRA','REGISTRAR_VENTA_CAJA')),
    CONSTRAINT ck_operaciones_estado CHECK (estado IN ('ACEPTADA','RECHAZADA','CONFLICTO','REQUIERE_ACEPTACION_PRECIO','PENDIENTE_PAGO','REGISTRADA_CENTRAL','CONCILIACION_REQUERIDA')),
    CONSTRAINT ck_operaciones_tiempo_conflicto CHECK ((codigo <> 'TIEMPO_INCOHERENTE' OR estado = 'CONFLICTO') AND (estado <> 'CONFLICTO' OR codigo = 'TIEMPO_INCOHERENTE')),
    CONSTRAINT ck_operaciones_aceptacion_fecha CHECK (aceptada_en IS NULL OR aceptada_en >= recibida_en)
);

(

    reserva_id uuid,
    sucursal_id uuid NOT NULL,
    cajero_id uuid,

    estado comercial.estado_venta NOT NULL DEFAULT 'PENDIENTE_PAGO'::comercial.estado_venta,
    subtotal numeric(12, 2) NOT NULL,
    descuento numeric(12, 2) NOT NULL DEFAULT 0,
    costo_entrega numeric(12, 2) NOT NULL DEFAULT 0,
    total numeric(12, 2) NOT NULL,
    creada_en timestamp with time zone NOT NULL DEFAULT now(),
    confirmada_en timestamp with time zone,
    clave_idempotencia uuid,
    hash_solicitud text COLLATE pg_catalog."default",
    adelanto_descontado numeric(12, 2) NOT NULL DEFAULT 0,
    expira_en timestamp with time zone,
    empresa_id uuid NOT NULL,
    CONSTRAINT ventas_pkey PRIMARY KEY (id),
    CONSTRAINT uq_ventas_id_empresa UNIQUE (id, empresa_id),
